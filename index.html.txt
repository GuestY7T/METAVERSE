  <script>
    /* STATE & DATA PERSISTENCE MANAGEMENT */
    let userAccount = JSON.parse(localStorage.getItem('game_user_v2')) || {
      username: 'Guest',
      coins: 500,
      xp: 1000,
      equipped: { skin: 'skin_guest' },
      savedPos: { x: 0, y: 0, z: 0 },
      cameraDist: 10
    };

    let friendsList = JSON.parse(localStorage.getItem('game_friends_v2')) || [
      { name: 'PlayerOne', online: true },
      { name: 'NovaRider', online: true },
      { name: 'PixelKing', online: false }
    ];

    let npcBots = [
      { name: 'PlayerOne', responses: ["Let's race to the jump pad!", "I love this green park!", "Check out my avatar!"] },
      { name: 'NovaRider', responses: ["Hey there!", "Scroll your mouse wheel to zoom in & out!", "Press 3 to flip!"] }
    ];

    let authMode = 'signup';

    function saveState() {
      localStorage.setItem('game_user_v2', JSON.stringify(userAccount));
      localStorage.setItem('game_friends_v2', JSON.stringify(friendsList));
      document.getElementById('user-coins').innerText = userAccount.coins;
      document.getElementById('user-xp').innerText = userAccount.xp;
    }

    /* UI TABS & MODALS */
    function switchTab(tab) {
      document.querySelectorAll('.chat-tab').forEach(t => t.classList.remove('active'));
      document.querySelectorAll('.panel-content').forEach(p => p.classList.remove('active'));
      
      if (tab === 'global') {
        document.querySelectorAll('.chat-tab')[0].classList.add('active');
        document.getElementById('panel-global').classList.add('active');
      } else if (tab === 'friends') {
        document.querySelectorAll('.chat-tab')[1].classList.add('active');
        document.getElementById('panel-friends').classList.add('active');
        renderFriends();
      } else if (tab === 'players') {
        document.querySelectorAll('.chat-tab')[2].classList.add('active');
        document.getElementById('panel-players').classList.add('active');
        renderPlayers();
      }
    }

    function openModal(id) {
      closeModals();
      document.getElementById(id).style.display = 'block';
      document.querySelectorAll('.nav-item').forEach(i => i.classList.remove('active'));
      if(id === 'modal-games') document.getElementById('nav-games').classList.add('active');
      if(id === 'modal-shop') document.getElementById('nav-shop').classList.add('active');
      if(id === 'modal-updates') document.getElementById('nav-updates').classList.add('active');
      if(id === 'modal-settings') document.getElementById('nav-settings').classList.add('active');
    }

    function closeModals() {
      document.querySelectorAll('.app-modal').forEach(m => m.style.display = 'none');
    }

    function showDashboard() {
      closeModals();
      document.getElementById('dashboard-overlay').style.display = 'block';
      document.querySelectorAll('.nav-item').forEach(i => i.classList.remove('active'));
      document.getElementById('nav-home').classList.add('active');
    }

    function buySkin(skinKey, cost) {
      if (userAccount.coins >= cost) {
        if (cost > 0 && userAccount.equipped.skin !== skinKey) userAccount.coins -= cost;
        userAccount.equipped.skin = skinKey;
        saveState();
        updateCharacterAppearance();
        alert('Equipped ' + skinKey + '!');
      } else {
        alert('Not enough coins!');
      }
    }

    /* FRIENDS & PLAYERS LIST */
    function renderFriends() {
      const list = document.getElementById('friends-list');
      list.innerHTML = '';
      friendsList.forEach(f => {
        const item = document.createElement('div');
        item.className = 'friend-item';
        item.innerHTML = `
          <div>
            <span class="status-dot ${f.online ? 'online' : 'offline'}"></span>
            <b>${f.name}</b>
          </div>
          <button class="btn-action" onclick="inviteFriend('${f.name}')">${f.online ? 'Invite' : 'Message'}</button>
        `;
        list.appendChild(item);
      });
    }

    function addFriend() {
      const input = document.getElementById('friend-search');
      if (input.value.trim() !== '') {
        friendsList.push({ name: input.value.trim(), online: true });
        input.value = '';
        saveState();
        renderFriends();
      }
    }

    function inviteFriend(name) {
      alert(`Sent party invitation to ${name}!`);
    }

    function renderPlayers() {
      const list = document.getElementById('players-list');
      list.innerHTML = '';
      const npcs = [{ name: userAccount.username, isSelf: true }, ...npcBots];
      npcs.forEach(p => {
        const item = document.createElement('div');
        item.className = 'player-item';
        item.innerHTML = `
          <div>
            <span class="status-dot online"></span>
            <b>${p.name}</b> ${p.isSelf ? '(You)' : ''}
          </div>
          ${!p.isSelf ? `<button class="btn-action" onclick="addFriendDirect('${p.name}')">+ Friend</button>` : ''}
        `;
        list.appendChild(item);
      });
    }

    function addFriendDirect(name) {
      if (!friendsList.some(f => f.name === name)) {
        friendsList.push({ name: name, online: true });
        saveState();
        alert(`Added ${name} to your friends list!`);
      }
    }

    /* CHAT & RESPONSIVE BOTS */
    const chatInput = document.getElementById('chat-input');
    const chatBox = document.getElementById('chat-messages');

    chatInput.addEventListener('keypress', (e) => {
      if (e.key === 'Enter' && chatInput.value.trim() !== '') {
        const userText = chatInput.value.trim();
        addChatMessage(userAccount.username, userText, false);
        chatInput.value = '';

        setTimeout(() => {
          const randomBot = npcBots[Math.floor(Math.random() * npcBots.length)];
          let reply = randomBot.responses[Math.floor(Math.random() * randomBot.responses.length)];
          
          if (userText.toLowerCase().includes('hi') || userText.toLowerCase().includes('hello')) {
            reply = `Hey ${userAccount.username}! Welcome!`;

    /* CONTROLS, ZOOM & PHYSICS */
    const keys = {};
    let playerPos = { x: userAccount.savedPos.x || 0, y: userAccount.savedPos.y || 0, z: userAccount.savedPos.z || 0 };
    let cameraDistance = userAccount.cameraDist || 10;
    let velocityY = 0;
    let isGrounded = true;
    let currentEmote = null;
    let emoteTimer = 0;

    // Roblox-Style Zoom In / Out controls via Wheel
    window.addEventListener('wheel', (e) => {
      if (e.deltaY > 0) {
        cameraDistance = Math.min(cameraDistance + 1.2, 25); // Zoom out
      } else {
        cameraDistance = Math.max(cameraDistance - 1.2, 3); // Zoom in
      }
      userAccount.cameraDist = cameraDistance;
      saveState();
    });

    function triggerEmote(emote) {
      currentEmote = emote;
      emoteTimer = 0;
      if (emote === 'flip') torsoMesh.rotation.x = 0;
    }

    window.addEventListener('keydown', (e) => {
      const activeTag = document.activeElement ? document.activeElement.tagName.toLowerCase() : '';
      if (activeTag === 'input' || activeTag === 'textarea') return;
      
      const key = e.key.toLowerCase();
      keys[key] = true;
      
      if ((key === ' ' || key === 'spacebar') && isGrounded) {
        velocityY = 0.25;
        isGrounded = false;
      }

      if (key === '1') triggerEmote('dance');
      if (key === '2') triggerEmote('wave');
      if (key === '3') triggerEmote('flip');
    });

    window.addEventListener('keyup', (e) => { keys[e.key.toLowerCase()] = false; });

    let saveCounter = 0;

    function animate() {
      requestAnimationFrame(animate);

      let moveX = 0, moveZ = 0;
      if (keys['w'] || keys['arrowup']) moveZ -= 1;
      if (keys['s'] || keys['arrowdown']) moveZ += 1;
      if (keys['a'] || keys['arrowleft']) moveX -= 1;
      if (keys['d'] || keys['arrowright']) moveX += 1;

      const isMoving = (moveX !== 0 || moveZ !== 0);
      if (isMoving && currentEmote) {
        currentEmote = null;
        torsoMesh.rotation.x = 0;
      }

      const currentSpeed = (keys['shift'] && isMoving) ? speed * 1.6 : speed;

      if (isMoving) {
        const len = Math.sqrt(moveX * moveX + moveZ * moveZ);
        playerPos.x += (moveX / len) * currentSpeed;
        playerPos.z += (moveZ / len) * currentSpeed;

        const targetAngle = Math.atan2(moveX, moveZ);
        let diff = targetAngle - playerGroup.rotation.y;
        while (diff < -Math.PI) diff += Math.PI * 2;
        while (diff > Math.PI) diff -= Math.PI * 2;
        playerGroup.rotation.y += diff * 0.2;
      }

      velocityY += -0.012;
      playerPos.y += velocityY;
      if (playerPos.y <= 0) {
        playerPos.y = 0;
        velocityY = 0;
        isGrounded = true;
      }

      // Jump Pad
      const dxPad = playerPos.x - jumpPad.position.x;
      const dzPad = playerPos.z - jumpPad.position.z;
      if (Math.sqrt(dxPad * dxPad + dzPad * dzPad) < 1.8 && playerPos.y <= 0.2) {
        velocityY = 0.55;
        isGrounded = false;
      }

      // Coins
      coins.forEach(coin => {
        if (coin.userData.active) {
          coin.rotation.z += 0.04;
          coin.position.y = coin.userData.baseY + Math.sin(Date.now() * 0.005) * 0.15;

          const dx = playerPos.x - coin.position.x;
          const dz = playerPos.z - coin.position.z;
          if (Math.sqrt(dx * dx + dz * dz) < 1.2 && Math.abs(playerPos.y - coin.position.y) < 1.5) {
            coin.userData.active = false;
            scene.remove(coin);
            userAccount.coins += 50;
            userAccount.xp += 25;
            saveState();
          }
        }
      });

      playerGroup.position.set(playerPos.x, playerPos.y, playerPos.z);

      // Wandering NPCs Update
      npcsInWorld.forEach(npc => {
        npc.group.position.x += npc.dirX * 0.03;
        npc.group.position.z += npc.dirZ * 0.03;

        const distFromBase = Math.sqrt(Math.pow(npc.group.position.x - npc.basePos.x, 2) + Math.pow(npc.group.position.z - npc.basePos.z, 2));
        if (distFromBase > 8) {
          npc.dirX *= -1;
          npc.dirZ *= -1;
        }
      });

      // Animations & Uncursed Backflip Math
      emoteTimer += 0.05;
      if (currentEmote === 'dance') {
        leftArmPivot.rotation.x = Math.sin(emoteTimer * 4) * 1.2;
        rightArmPivot.rotation.x = -Math.sin(emoteTimer * 4) * 1.2;
        playerGroup.rotation.y += 0.05;
      } else if (currentEmote === 'wave') {
        rightArmPivot.rotation.x = -2.5;
        rightArmPivot.rotation.z = Math.sin(emoteTimer * 6) * 0.4;
      } else if (currentEmote === 'flip') {
        // Uncursed flip: clean full 360 degree rotation pitch reset
        const flipProgress = emoteTimer * 4;
        if (flipProgress <= Math.PI * 2) {
          torsoMesh.rotation.x = -flipProgress;
        } else {
          torsoMesh.rotation.x = 0;
          currentEmote = null;
        }
      } else if (!isGrounded) {
        leftArmPivot.rotation.x = rightArmPivot.rotation.x = -1.2;
        leftLegPivot.rotation.x = 0.3;
        rightLegPivot.rotation.x = -0.3;
      } else if (isMoving) {
        leftArmPivot.rotation.x = Math.sin(emoteTimer * 2) * 0.7;
        rightArmPivot.rotation.x = -Math.sin(emoteTimer * 2) * 0.7;
        leftLegPivot.rotation.x = -Math.sin(emoteTimer * 2) * 0.7;
        rightLegPivot.rotation.x = Math.sin(emoteTimer * 2) * 0.7;
      } else {
        rightArmPivot.rotation.z = 0;
        headMesh.position.y = 2.15 + Math.sin(emoteTimer) * 0.04;
        leftArmPivot.rotation.x = rightArmPivot.rotation.x = 0;
        leftLegPivot.rotation.x = rightLegPivot.rotation.x = 0;
      }

      // Roblox Dynamic Camera Follow & Zoom Distance
      camera.position.x += (playerPos.x - camera.position.x) * 0.1;
      camera.position.y += ((playerPos.y + (cameraDistance * 0.55)) - camera.position.y) * 0.1;
      camera.position.z += ((playerPos.z + cameraDistance) - camera.position.z) * 0.1;
      camera.lookAt(playerPos.x, playerPos.y + 1.5, playerPos.z);

      // Save player position every 60 frames (~1s)
      saveCounter++;
      if (saveCounter > 60) {
        userAccount.savedPos = { x: playerPos.x, y: playerPos.y, z: playerPos.z };
        saveState();
        saveCounter = 0;
      }

      renderer.render(scene, camera);
    }

    window.addEventListener('resize', () => {
      camera.aspect = window.innerWidth / window.innerHeight;
      camera.updateProjectionMatrix();
      renderer.setSize(window.innerWidth, window.innerHeight);
    });

    saveState();
    updateCharacterAppearance();
    animate();
